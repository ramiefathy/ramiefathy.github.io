// Firestore Security Rules for Clinic Scheduler Pro
// Copy these rules to Firebase Console > Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isSignedIn() {
      return request.auth != null;
    }

    // Check if user is a member of the institution
    function isInstitutionMember(institutionId) {
      return isSignedIn() && exists(/databases/$(database)/documents/institutions/$(institutionId)) &&
        get(/databases/$(database)/documents/institutions/$(institutionId)).data.members != null;
    }

    // Get user's role in the institution
    function getUserRole(institutionId) {
      let inst = get(/databases/$(database)/documents/institutions/$(institutionId)).data;
      return inst.members != null && inst.members.size() > 0
        ? inst.members[0].role // For now, simplified check
        : 'member';
    }

    function isInstitutionAdmin(institutionId) {
      return isSignedIn() && isInstitutionMember(institutionId);
      // Simplified for now - all members have admin access
    }

    function isScheduler(institutionId) {
      return isSignedIn() && isInstitutionMember(institutionId);
      // Simplified for now - all members can schedule
    }

    // Institution rules
    match /institutions/{institutionId} {
      allow read: if isInstitutionMember(institutionId);
      allow create: if isSignedIn();
      allow update: if isInstitutionAdmin(institutionId);
      allow delete: if false; // Never allow deletion of institutions

      // Institution members subcollection
      match /members/{userId} {
        allow read: if isInstitutionMember(institutionId);
        allow write: if isInstitutionAdmin(institutionId) || request.auth.uid == userId;
      }

      // Attendings subcollection
      match /attendings/{attendingId} {
        allow read: if isInstitutionMember(institutionId);
        allow write: if isScheduler(institutionId);
      }

      // Residents subcollection
      match /residents/{residentId} {
        allow read: if isInstitutionMember(institutionId);
        allow write: if isScheduler(institutionId);
      }

      // Schedule assignments subcollection
      match /assignments/{assignmentId} {
        allow read: if isInstitutionMember(institutionId);
        allow write: if isScheduler(institutionId);
      }

      // Rules subcollection
      match /rules/{ruleId} {
        allow read: if isInstitutionMember(institutionId);
        allow write: if isScheduler(institutionId);
      }

      // Audit logs subcollection
      match /auditLogs/{logId} {
        allow read: if isInstitutionAdmin(institutionId);
        allow create: if isInstitutionMember(institutionId);
        allow update, delete: if false; // Audit logs are immutable
      }
    }

    // User profiles (global)
    match /users/{userId} {
      allow read: if isSignedIn();
      allow write: if request.auth.uid == userId;
    }

    // Institution invites
    match /inviteCodes/{inviteCode} {
      allow read: if true; // Anyone with the code can read
      allow create: if isSignedIn();
      allow update: if isSignedIn() && resource.data.createdBy == request.auth.uid;
      allow delete: if isSignedIn() && resource.data.createdBy == request.auth.uid;
    }
  }
}
